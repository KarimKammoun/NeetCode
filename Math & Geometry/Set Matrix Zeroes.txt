class Solution:
    def setZeroes(self, matrix: List[List[int]]) -> None:
        n=len(matrix)
        m=len(matrix[0])
        
        list_x=set()
        list_y=set()

        for i in range(n):
            for j in range(m):
                if  matrix[i][j]==0:
                    list_x.add(i)
                    list_y.add(j)

        for i in list_x:
            for j in range(m):
                matrix[i][j]=0

        for i in list_y:
            for j in range(n):
                matrix[j][i]=0
